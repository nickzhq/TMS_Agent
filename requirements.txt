# EstimatePrice Agent - Python 依赖
# 用户自行安装：pip install -r requirements.txt（AI 不自动装包）
# 技术选型见 docs/architecture.md 第三节 / record.md 第十节

langgraph>=0.2          # 编排框架（图状态机 + interrupt + checkpoint）
langchain-core          # LangGraph 依赖（State / 消息基类）
openai>=1.0             # 裸 OpenAI 兼容 SDK（切 base_url 调 Qwen/DeepSeek/自定义）
mem0ai                  # 记忆层（仅用户偏好）
pydantic>=2.0           # 配置模型 + State 校验
pyyaml                  # config.yaml 解析
python-dotenv           # 环境变量（DB 连接串 / API key，绝不入库）
mcp                     # 高德 MCP 客户端（maps_geo）；DB 已改 pyodbc 直连
pyodbc                  # DB 直连 SQL Server（跑标准 T-SQL，含 CTE/DECLARE/多语句）

# 界面 / API / 持久化（2026-09-30 形态调整：网页应用）
fastapi                 # API 层（REST 会话 + WebSocket 流式/HITL）
uvicorn                 # ASGI 服务器（跑 FastAPI）
chainlit                # 前端阶段1（对话 UI，纯 Python，MVP）
langgraph-checkpoint-sqlite  # 会话持久化 Checkpointer（SQLite 起步）
websockets              # WebSocket（流式 + HITL 双向）
aiosqlite               # 异步 SQLite（FastAPI 异步）
