# Python dependencies
# Install yourself: pip install -r requirements.txt (the assistant does not auto-install)

langgraph>=0.2          # orchestration (graph state machine + interrupt + checkpoint)
langchain-core          # LangGraph dependency (State / message base classes)
openai>=1.0             # bare OpenAI-compatible SDK (switch base_url for any provider)
mem0ai                  # memory layer (user preferences only)
qdrant-client           # mem0 vector store: Qdrant local (embedded) mode with on-disk persistence
pydantic>=2.0           # config models + state validation
pyyaml                  # config.yaml parsing
python-dotenv           # env vars (DB connection string / API key, never committed)
mcp                     # map-service MCP client (geocoding); DB uses pyodbc directly
pyodbc                  # direct SQL Server connection (runs standard T-SQL: CTE/DECLARE/multi-statement)

# Web UI / API / persistence
fastapi                 # API layer (REST sessions + WebSocket streaming / HITL)
uvicorn                 # ASGI server (runs FastAPI)
chainlit                # chat UI (pure Python, MVP)
langgraph-checkpoint-sqlite  # session persistence checkpointer (SQLite, dev/tests)
langgraph-checkpoint-postgres  # Postgres checkpointer (P8 batch-2, multi-user production)
psycopg[binary,pool]    # psycopg v3 + async pool (Postgres driver: checkpointer/sessions/auth) (P8 batch-2)
websockets              # WebSocket (streaming + HITL bidirectional)
aiosqlite               # async SQLite (FastAPI async)
httpx                   # frontend -> backend HTTP client (login + sessions REST) (P8)
pyjwt                   # backend JWT signing/verification for real auth (P8)

# Notes:
# - pyodbc requires the system-level ODBC Driver for SQL Server (Microsoft official driver)
# - SQLite is built into Python (sqlite3); no extra install needed
# - multi-user production: swap in langgraph-checkpoint-postgres + PostgreSQL
# - password hashing uses the stdlib (hashlib.pbkdf2_hmac, PBKDF2-HMAC-SHA256) -> no passlib/bcrypt needed
# - the knowledge base is an external Go service (called via REST/MCP), not a Python package
